# --- 2. USAGE LIMITS & RESET ENGINE ---
def get_today_date():
    return datetime.now().strftime("%Y-%m-%d")

def load_usage():
    if os.path.exists(USAGE_FILE):
        try:
            with open(USAGE_FILE, "r", encoding="utf-8") as f:
                data = json.load(f)
                if data.get("date") == get_today_date():
                    return data
        except Exception:
            pass
    return {
        "date": get_today_date(),
        "pro_used": 0,
        "max_used": 0,
        "img_used": 0,
        "tier": "Free"
    }

def save_usage(usage_data):
    try:
        with open(USAGE_FILE, "w", encoding="utf-8") as f:
            json.dump(usage_data, f, ensure_ascii=False, indent=2)
    except Exception:
        pass

if "usage" not in st.session_state:
    st.session_state.usage = load_usage()

if st.session_state.usage["date"] != get_today_date():
    st.session_state.usage = {
        "date": get_today_date(),
        "pro_used": 0,
        "max_used": 0,
        "img_used": 0,
        "tier": st.session_state.usage.get("tier", "Free")
    }
    save_usage(st.session_state.usage)

# --- 3. LONG-TERM MEMORY ENGINE ---
def load_long_term_memory():

# --- 9. INPUT & ACTION HANDLER ---
chat_input = st.chat_input("Ask Luxurity AI or command an app open...")
active_prompt = st.session_state.quick_prompt or chat_input
st.session_state.quick_prompt = None

if active_prompt:
    if not api_key:
        st.warning("⚠️ Please enter your Gemini API Key in the left sidebar!")
    else:
        # Check for smart app-opening commands
        lower_prompt = active_prompt.lower()
        if "open youtube" in lower_prompt:
            webbrowser.open("https://www.youtube.com")
            action_reply = "🚀 Opening YouTube for you right now!"
        elif "open spotify" in lower_prompt:
            webbrowser.open("https://open.spotify.com")
            action_reply = "🎵 Opening Spotify for you right now!"
        elif "open amazon music" in lower_prompt:
            webbrowser.open("https://music.amazon.com")
            action_reply = "🎶 Opening Amazon Music for you right now!"
        else:
            action_reply = None

        st.session_state.messages.append({"role": "user", "content": active_prompt})
        with st.chat_message("user"):
            st.markdown(active_prompt)

        with st.chat_message("assistant"):
            if action_reply:
                st.markdown(action_reply)
                st.session_state.messages.append({"role": "assistant", "content": action_reply})
                save_memory(st.session_state.messages)
            else:
                placeholder = st.empty()
                full_reply = ""

                try:
                    client = genai.Client(api_key=api_key)
                    tools = [{"google_search": {}}] if enable_search else None

                    # Inject long-term memory into system prompt context
                    ltm_context = "User's Saved Long-Term Memories:\n" + "\n".join([f"- {m}" for m in st.session_state.long_term_memory]) if st.session_state.long_term_memory else ""
                    combined_instruction = f"{base_instruction}\n\n{ltm_context}"

                    contents = []
                    for m in st.session_state.messages[:-1]:
                        role = "user" if m["role"] == "user" else "model"
                        contents.append(types.Content(role=role, parts=[types.Part.from_text(text=m["content"])]))

                    contents.append(types.Content(role="user", parts=[types.Part.from_text(text=active_prompt)]))

                    response = client.models.generate_content_stream(
                        model="gemini-2.5-flash",
                        contents=contents,
                        config=types.GenerateContentConfig(
                            system_instruction=combined_instruction,
                            tools=tools
                        )
                    )

                    for chunk in response:
                        if chunk.text:
                            full_reply += chunk.text
                            placeholder.markdown(full_reply + " ▌")

                    placeholder.markdown(full_reply)
                    st.session_state.messages.append({"role": "assistant", "content": full_reply})
                    save_memory(st.session_state.messages)

                except Exception as e:
                    st.session_state.messages.pop()
                    st.error(f"❌ Error: {e}")
